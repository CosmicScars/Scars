import wget  # Si no lo tienes: !pip install wget
import healpy as hp
import matplotlib.pyplot as plt
import numpy as np
from matplotlib.colors import LinearSegmentedColormap
from google.colab import drive
drive.mount('/content/drive')

# 1. Cargar el mapa I_Stokes y convertir a μK (sin máscara inicial)
cmb = hp.read_map("/content/drive/MyDrive/CCC/HFI_SkyMap_143_2048_R3.01_full.fits", field=0) * 1e6  # K_CMB → μK

# 2. Definir colormap estilo "Planck Legacy" (azul-blanco-rojo)
planck_cmap = LinearSegmentedColormap.from_list("planck", ["#00007F", "#0000FF", "#FFFFFF", "#FF0000", "#7F0000"])

# 3. Aplicar filtro pasa-alto para eliminar gradientes a gran escala
lmax = 1024  # Armónicos esféricos hasta l=1024
alm = hp.map2alm(cmb, lmax=lmax)
alm_filtered = alm.copy()
alm_filtered[0] = 0  # Eliminar monopolo
for l in range(1, 20):  # Eliminar dipolo (l=1) y modos a gran escala (l<20)
    alm_filtered[hp.sphtfunc.Alm.getidx(lmax, l, 0)] = 0
cmb_filtered = hp.alm2map(alm_filtered, hp.get_nside(cmb))

# 4. Visualización profesional (ajuste manual de escala)
plt.figure(figsize=(18, 12))
hp.mollview(
    cmb_filtered,
    title="CMB 143 GHz: Real Anisotropies (High-Pass Filtering, l≥20)",
    unit="μK",
    min=-200,  # Ajustado manualmente para resaltar patrones
    max=200,
    cmap=planck_cmap,
    remove_dip=True,
    remove_mono=True,
    hold=True
)

# 5. Añadir grid galáctico para referencia (opcional)
hp.graticule(verbose=False, color="gray", alpha=0.3)

# 6. Guardar en alta resolución
plt.savefig("/content/drive/MyDrive/CCC/cmb_real_pro.png", dpi=600, bbox_inches='tight')
plt.close()
